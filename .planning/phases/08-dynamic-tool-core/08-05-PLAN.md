**2. frontend/src/lib/api.ts** — 添加 confirmToolCreation API 函数：

```typescript
export async function confirmToolCreation(
  sessionId: string,
  confirmationId: string,
  approved: boolean,
  persistence: string = "session",
  extraDirs: string[] = [],
): Promise<void> {
  const response = await fetch(
    `/api/sessions/${encodeURIComponent(sessionId)}/confirm-tool-creation`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        confirmation_id: confirmationId,
        approved,
        persistence,
        extra_dirs: extraDirs,
      }),
    }
  );
  return handleResponse<void>(response);
}
```
